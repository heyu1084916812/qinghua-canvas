  // 「递归」是故意的——上游的显示快照变化要传到下游
  inEdgesHash: string
  // 三者拼接后的总指纹
  combined: string
}

export function computeFingerprint(
  node: NodeSnapshot,
  params: NodeData,
  upstreamFingerprints: NodeFingerprint[]

```ts
// domain/canvas/model/runRecord.ts
// RunRecord = 一次成功/失败生成的完整记录，对应产品文档 §6.21 节点版本历史
// 「从不删除」是硬性要求：每个生成类节点保留全部版本，可随时回溯与恢复
export interface RunRecord {
  id: string
  nodeId: string
  projectId: string
  // 版本号，节点内自增（v1 / v2 / v3 …）
  version: number
  createdAt: number
  // 本次执行的状态
  status: 'succeeded' | 'failed' | 'canceled' | 'interrupted'
  // 输入快照：本次执行冻结的入边素材与参数（恢复版本时使用）
  inputs: NodeInput[]
  params: NodeData
  // 输出：本次执行产出的素材哈希列表
  outputHashes: string[]
  // 本次执行产生的指纹——这是「LiveFingerprint」的载体
  // 节点当前生效的 LiveFingerprint = 该节点最新一条 status='succeeded' 的 RunRecord.fingerprint
  fingerprint: NodeFingerprint
  // 溯源：来源任务 id，可反查日志面板的完整请求/响应/耗时/成本
  taskId: string
  durationMs: number
  cost?: number
}

export function createRunRecord(args: {
  nodeId: string
  version: number
  inputs: NodeInput[]
  params: NodeData
  outputHashes: string[]
  fingerprint: NodeFingerprint
  taskId: string
}): RunRecord
```

**LiveFingerprint 的存取规则**：不单独建表，也不存在节点行内——它派生自 RunRecord。节点当前生效的 LiveFingerprint = 该节点 `version` 最大且 `status='succeeded'` 那条 RunRecord 的 `fingerprint`。「恢复旧版本」= 追加一条新 RunRecord（版本自增、沿用旧版的 inputs/params/outputHashes），旧版本记录保留，因此 LiveFingerprint 随之指向新版。

> **2026-10-06 端口补一条 `vision`（对账清单 #176，情绪「识别人脸」改成先在本机认）**：
> 参照 VOZEB-PRO，人脸检测不再只靠「问对话模型」，而是加了一条**本机**路

> 三条契约值得记下来：
>
> - **不出网**：所以它与 `network` 是两个端口。静态文件（`vision_bundle.js` / wasm / tflite）

> 是 `/canvas/<id>`（没有结尾斜杠）⇒ 解析成 `/canvas/mediapipe/...` ⇒ 被 dev server 的
> SPA 回退成 `index.html` 200 ⇒ worker 拿到一坨 HTML，**建得起来、跑不起来**。
> 要命的是这不会报错：回退路径会把整条流程照常跑绿。修法是按 `import.meta.env.BASE_URL`
> 取根路径。**通用教训：静态资源路径一律以 BASE_URL 为根，不要以当前文档为根**；
> 以及「有回退的功能，必须有一条断言证明主路真的被走了」。

```ts
export interface CommandResult {
  patches: Patch[]          // 正向补丁，用于渲染
  inverse: Patch[]          // 逆向补丁，用于撤销
  transaction: TransactionBoundary
  persist: PersistPlan      // { tables, upserts, deletes }
}

export type TransactionBoundary =
  | { mode: 'standalone'; label: string }              // 立即成事务
  | { mode: 'coalesce'; groupId: string; label: string } // 与同 groupId 合并（拖动 / 缩放）
  | { mode: 'silent' }                                  // 瞬态，不进撤销栈（拖线预览等）
  | { mode: 'multi-step'; planId: string; label: string } // 多步事务：拓扑生成期间所有 node.create + resultGroup.create 合并为 1 个撤销单元

// multi-step 的语义要点：
// 1. planId 相同的连续 Command 共享一个撤销步骤，undo 一次性回退整次拓扑生成
// 2. 每个 Command 仍独立 dispatch、独立 apply patches、独立持久化——保证渲染与持久化的实时性
// 3. planId 由 runPlan.execute 创建，runPlan.cancel 或 runPlan 完成时关闭
// 4. 历史栈只入栈 1 次，且是 lazy 的——首个携带该 planId 的 Command 真正 dispatch 时才创建栈项，
//    后续同 planId Command 合并到该栈项。不在 buildRunPlan 时预先入栈：
//    计划可能校验失败、用户可能在第一个节点完成前就取消，预先入栈会产生空事务与幽灵撤销步骤
// 5. multi-step 不允许嵌套（嵌套会改写为外层 planId，嵌套层入栈单独）
// 6. 取消时的语义：已 dispatch 的 Command 保留（不回滚），仅停止后续 Command 的产生。
//    整次拓扑生成仍是一个撤销单元——undo 会一次性回退到计划开始前的状态
```

**唯一入口原则**：图数据的每一次变更都经过 `dispatch(command)`。视口、选中、UI 浮层属于瞬态状态，走独立 API（`setSelection` / `setViewport`），不进 dispatch、不进撤销栈、不触发持久化。（`stale.mark` / `stale.clear` 曾走命令且不进 undo 栈，已随陈旧标记于 2026-09-17 下线。）

`createStore` 是工厂而非模块级单例：每个工作台页面挂载时调用一次，注入本工作台的私有切片与命令。共享切片随工厂闭包被引用，工作台私有切片随路由生命周期创建销毁。

### 4.5 canvas：渲染层与节点注册表 · ✅ 已实现

> **注册表拆成两半**（本次修订。原设计把行为与渲染混在 `workbenches/canvas/nodes/registry.ts`，存在依赖方向矛盾）
>
> 命令层在 `state`（`node.create` 需要 `createDefaultData`、`container.reparent` 需要 `accepts`）、执行层在 `features`（需要 `collectInputs` / `generate`）、domain 自身（`canConnect` 需要 `accepts`）都要读节点规格；而依赖方向禁止 `state` → `workbenches`。规格若留在渲染层，命令层只能写死 `switch (type)`——「新增一个节点类型只改 3 个文件」的承诺随之失效。
>
> | 内容 | 位置 | 谁依赖它 |
> | --- | --- | --- |
> | `NodeSpec`：`label` `sizing` `ports` `accepts` `createDefaultData` `collectInputs` `generate` `toRunRequest` | `domain/canvas/nodeSpecs/` | `state` 命令层、`features` 执行层、`domain` 规则层 |
> | `NodeViewBinding`：`View` `Panel` | `workbenches/canvas/nodes/<name>/` | 仅渲染层 |
>
> `NodeDefinition = NodeSpec & NodeViewBinding` 只在渲染装配处组合，不进入 domain。两者以 `NodeType` 联合类型作为唯一契约，缺一即视为未注册完成。

```ts
   * 定义「点生成时这个节点跑几次 API、结果落哪里」。
   * 返回 null 表示该类型不可生成（对比节点）。
   */

  /** 需要准备的空槽位数；placement = 'downstream-slots' 时通常等于 calls.length */
  slotCount: number

 *
 * | 类型   | calls                        | placement          | slotCount              |
 * |--------|------------------------------|--------------------|------------------------|
 * | 提示词 | [ 文本请求 ×1 ]               | 'self'             | 0（覆盖自身文本框）     |
 * | 图片   | [ 生图请求 × count ]          | 'downstream-slots' | count                  |
 * | 视频   | [ 视频请求 × count ]          | 'downstream-slots' | count                  |
 * | 批量   | [ 每素材每 count 一次请求 ]    | 'downstream-slots' | count × 内部素材数      |
 * | 分组   | [ 所有素材一次请求 ]          | 'downstream-slots' | 创作面板「生成数量」参数 |
 * | 对比   | null（不可生成）              | —                  | —                      |
```

### 4.7 pages：页面装配层 · ✅ 已实现

页面容器是唯一允许「把各层接起来」的地方。它不实现业务规则，只做四件接线工作。以 `CanvasPage` 为例：

```ts
// pages/CanvasPage/useCanvasPageEvents.ts
// ① 视图事件 → 命令：节点视图 emit 上来的事件在这里翻译成 Command
export function useCanvasPageEvents(): {
  onNodeEvent(nodeId: string, event: NodeViewEvent): void
  onMenuAction(nodeId: string, action: NodeMenuAction): void
}
```

```
② 创建 writeBack 闭包（注入 dispatch）：
     const writeBack = (commands: Command[], tx?: TransactionBoundary) => {
       for (const c of commands) {
         if (c.kind === 'node.runRecord.append') noteRecord(c.record)  // 成功记录即新基线，同步记账
         dispatch(c, tx)
       }
     }
   —— runEngine 传来的 transaction 原样透传，装配层不判断也不改写；
      陈旧基线必须**同步**更新（不能让 React 的批处理拖后一帧，否则本帧对账拿旧基线比对）

③ runPlan 结束后的收尾（在 useExecution 的 then / finally 里）：
     · 把 summary.records 落库（经 ② 的 writeBack → dispatch node.runRecord.append）
     · 成功节点由对账自动清陈旧（基线已更新 → 进入 fresh 集合 → dispatch stale.clear）
     · 失败节点保留陈旧标记，等用户手动处理

④ 启动时（app/bootstrap）按顺序执行：
     migrateDatabase()
      → recoverInterrupted()      把中断的任务与产物节点改为失败态，不自动重发请求
       → 陈旧对账         中断可能让部分节点跑了一半，指纹必须重算一遍
                           （实现上由执行宿主挂载时读回 runRecords 重建成功基线完成，见 §4.4）
   顺序不可颠倒：先恢复再算陈旧，否则会拿残缺的显示快照去比对，误标一批陈旧节点
```

装配层的边界：**只接线，不放规则**。判断合法性是 `domain/rules` 的事，调度是 `runEngine` 的事，渲染是 `workbenches` 的事。装配层出现 `if` 判断业务条件，说明规则放错了层。

**导航出口同样走事件，不走命令**：工作台 UI 层不认识路由（本节边界）。节点 / 面板需要「离开本工作台」时，emit 一个**宿主导航语义事件**（`NodeViewEvent` / `PanelEvent` 各有一个 `openSettings`），由 `pages` 层翻译成 `useNavigate('/settings', { state: { from } })`。这样 `workbenches/` 不必 import `react-router`，路由依赖不会渗进工作台；画布与漫画剧两页共用同一条出口语义，`ComicPage` 与 `CanvasPage` 各自接线。设置页据 `state.from` 决定返回目标与文案（画布 / 漫画剧 / 首页），使「进来—回去」闭环。
